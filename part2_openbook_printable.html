⚙️ PART A: HARDWARE REFERENCE (หมวดฮาร์ดแวร์ — โครงสร้างชิป พิน รีจิสเตอร์ & วงจร)

1.1 สถาปัตยกรรมและโครงสร้างฮาร์ดแวร์ ATmega328P RISC MCU HARDWARE 1
  • สถาปัตยกรรมบัส: Harvard Architecture (แยกบัส Program Flash ROM 32KB และ Data SRAM 2KB ออกจากกัน ดึงคำสั่งได้ใน 1 คาบ Clock)
  • รีจิสเตอร์ทำงานทั่วไป: 32 General Purpose Working Registers ($R_0$ ถึง $R_{31}$) ขนาด 8 บิต โดยที่ $R_{16}-R_{31}$ ใช้คำสั่ง LDI ได้
  • คู่อ้างอิงที่อยู่ 16 บิต (Pointer Registers): X ($R_{27}:R_{26}$), Y ($R_{29}:R_{28}$), Z ($R_{31}:R_{30}$)
  • Program Counter (PC): 14 บิต ชี้ตำแหน่งแอดเดรสคำสั่งถัดไปใน Flash ROM
  • Stack Pointer (SP): 16 บิต ประกอบด้วย SPH (High Byte) และ SPL (Low Byte) ชี้แอดเดรสบน SRAM สำหรับเก็บ Return Address ตอน CALL และ PUSH/POP

🔹 แม่แบบที่ 2: การวนลูปนับเลข 1 ถึง 10 ออกพอร์ต PORTB

OUT DDRB, R16           ; ตั้ง PORTB เป็นเอาต์พุต
LDI R17, 1              ; ค่าเริ่มต้นนับ = 1

COUNT_LOOP:
    OUT PORTB, R17      ; ส่งค่านับออก PORTB
    CPI R17, 10         ; เช็คว่าครบ 10 หรือยัง
    BREQ LOOP_DONE      ; ถ้าครบ 10 ให้กระโดดออก
    INC R17             ; เพิ่มค่าทีละ 1
    RJMP COUNT_LOOP
LOOP_DONE:
🔹 แม่แบบที่ 3: อัลกอริทึมแปลง Hex 0-99 เป็น BCD 2 หลักด้วยวิธีลบซ้ำด้วย 10 
LDI R16, 0x45           ; ค่าอินพุต Hex 0x45 = 69 ทศนิยม
CLR R17                 ; R17 เก็บหลักสิบ (Tens)

BCD_CONVERT:
    CPI R16, 10         ; เปรียบเทียบค่าที่เหลือกับ 10
    BRLO BCD_DONE       ; นอกนั้นถ้าน้อยกว่า 10 จบลูป
    SUBI R16, 10        ; ลบออกทีละ 10
    INC R17             ; หลักสิบเพิ่มขึ้นทีละ 1
    RJMP BCD_CONVERT

BCD_DONE:
    SWAP R17            ; สลับหลักสิบไปไว้ที่ 4 บิตสูง (High Nibble)
    OR R17, R16         ; รวมหลักหน่วย (Low Nibble) ได้ BCD สมบูรณ์
    OUT PORTB, R17      ; ส่ง BCD ออก PORTB
🔹 แม่แบบที่ 4: การอ่านปุ่มกด Active Low สวิตช์ PINC บิต 0 ควบคุมไฟ LED PORTB บิต 5 
SBI DDRB, 5             ; ตั้ง PB5 เป็นเอาต์พุต (LED)
CBI DDRC, 0             ; ตั้ง PC0 เป็นอินพุต (Switch)
SBI PORTC, 0            ; เปิด Pull-up บน PC0

READ_SWITCH:
    SBIC PINC, 0        ; ข้ามคำสั่งถัดไปถ้าสวิตช์ PC0 ถูกกดลงดิน (เป็น 0)
    RJMP TURN_OFF       ; ถ้ายังไม่ได้กด (เป็น 1) ดับไฟ
TURN_ON:
    SBI PORTB, 5        ; ติดไฟ LED
    RJMP READ_SWITCH
TURN_OFF:
    CBI PORTB, 5        ; ดับไฟ LED

; ฟังก์ชันย่อยสร้างความหน่วงเวลาประมาณ 1 มิลลิวินาที (1ms) ที่คริสตัล 16MHz
DELAY_1MS:
    LDI R20, 20
L1: LDI R21, 200
L2: DEC R21             ; 1 Clock Cycle
    BRNE L2             ; 2 Clock Cycles (รวมลูปใน 3 Cycles x 200 = 600 Cycles)

📂 1. Lab 3 Code: AssemblyBlink1.asm (ควบคุมไฟ LED ติดดับพร้อมกันบน PORTB และ PORTD) 
;====================================================================
; Title: AssemblyBlink1.asm
; Created: 11/22/2013
; Author: khatathapswatdipisal
; Description: Blink all LEDs on PORTB and PORTD simultaneously
;====================================================================

.nolist
.include "m328pdef.inc"
.list

; Start at main after reset
rjmp main

main:
    ldi r16, 0xFF       ; Load 0xFF into R16 (all bits set to 1)
    out DDRB, r16       ; Configure Port B pins as Outputs
    out DDRD, r16       ; Configure Port D pins as Outputs
    ldi r16, 0x00       ; Clear R16 to 0x00

loop:
    com r16             ; Complement R16 (toggle bits between 0x00 and 0xFF)
    out PORTB, r16      ; Write R16 to Port B
    out PORTD, r16      ; Write R16 to Port D
    call MY_DELAY       ; Delay for simulation timing
    rjmp loop           ; Loop indefinitely

; Delay Subroutine
MY_DELAY:
    ldi r20, 8
L1:
    ldi r21, 200
L2:
    ldi r22, 250
L3:
    dec r22             ; Decrement R22
    brne L3             ; Loop if R22 != 0
    dec r21             ; Decrement R21
    brne L2             ; Loop if R21 != 0
    dec r20             ; Decrement R20
    brne L1             ; Loop if R20 != 0
    ret                 ; Return from subroutine
📂 2. Lab 3 Code: AssemblyBlink1_modified.asm (ควบคุมไฟ LED ติดสลับกัน PORTB ติดเมื่อ PORTD ดับ) 
;====================================================================
; Title: AssemblyBlink1_modified.asm
; Created: 11/22/2013 (Modified: 2026-07-19)
; Author: khatathapswatdipisal
; Description: Alternate blinking: Port B ON when Port D is OFF
;====================================================================

.nolist
.include "m328pdef.inc"
.list

; Start at main after reset
rjmp main

main:
    ldi r16, 0xFF       ; Load 0xFF into R16 (all bits set to 1)
    out DDRB, r16       ; Configure Port B pins as Outputs
    out DDRD, r16       ; Configure Port D pins as Outputs
    ldi r16, 0x00       ; Clear R16 to 0x00

loop:
    com r16             ; Complement R16 (toggle R16 between 0x00 and 0xFF)
    out PORTB, r16      ; Write R16 directly to Port B (PB0-PB7)
    
    mov r17, r16        ; Copy R16 to R17
    com r17             ; Complement R17 (so R17 is always the opposite of R16)
    out PORTD, r17      ; Write R17 to Port D (PD0-PD7)
    
    call MY_DELAY       ; Delay for simulation timing
    rjmp loop           ; Loop indefinitely

; Delay Subroutine
MY_DELAY:
    ldi r20, 8
L1:
    ldi r21, 200
L2:
    ldi r22, 250
L3:
    dec r22             ; Decrement R22
    brne L3             ; Loop if R22 != 0
    dec r21             ; Decrement R21
    brne L2             ; Loop if R21 != 0
    dec r20             ; Decrement R20
    brne L1             ; Loop if R20 != 0
    ret                 ; Return from subroutine

        {
            // Crude software delay
        }
    }
    
    PORTD = 0x00;       // Set Port D output low
    PORTB = 0x00;       // Set Port B output low
    
    while (1)
    {
        PORTD = 0xFF;                   // Turn all Port D LEDs ON
        PORTB = PORTB | 0b00100000;     // Turn PB5 LED ON

📂 4. AVR Exercise 5-8 Code: Hex to BCD Conversion (แปลง Hex 0xFD เป็น BCD 3 หลัก) 
.INCLUDE "M328PDEF.INC"
.ORG 0x0000

; โปรแกรมแปลงเลข Hex ใน R16 (เช่น 0xFD = 253 ทศนิยม) เป็น BCD 3 หลัก (หลักร้อย, หลักสิบ, หลักหน่วย)
LDI R16, 0xFD       ; 0xFD = 253 ทศนิยม
CLR R17             ; R17 เก็บหลักสิบ (Tens)
CLR R18             ; R18 เก็บหลักร้อย (Hundreds)

CONVERT_100:
    CPI R16, 100
    BRLO CONVERT_10
    SUBI R16, 100   ; ลบออกทีละ 100
    INC R18         ; เพิ่มหลักร้อย
    RJMP CONVERT_100

CONVERT_10:
    CPI R16, 10
    BRLO FINISH
    SUBI R16, 10    ; ลบออกทีละ 10
    INC R17         ; เพิ่มหลักสิบ
    RJMP CONVERT_10

FINISH:
    ; ผลลัพธ์: R18 = หลักร้อย (2), R17 = หลักสิบ (5), R16 = หลักหน่วย (3)
    OUT PORTB, R17  ; ส่งผลลัพธ์ออก PORTB
📂 5. AVR Exercise Code: Summation Loop (คำนวณผลรวม S = N + (N-1) + ... + 1)
